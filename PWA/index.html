function generate(){
  const raw = $('htmlInput').value;
  if (!raw.trim()){ alert('Cole seu index.html primeiro.'); return; }

  const doc = parseHTML(raw);

  const title = (doc.title || 'Meu App PWA').trim();
  const shortName = title.slice(0, 12);
  const lang = doc.documentElement.getAttribute('lang') || 'pt-BR';
  const themeColor = getMeta(doc, 'theme-color') || '#004aad';
  const backgroundColor = getMeta(doc, 'background-color') || themeColor;
  const description = getMeta(doc, 'description') || '';
  const faviconHref = detectFavicon(doc); // pode ser vazio

  const iconsPath = $('iconsPath').value.trim() || 'icons/icon';
  const iconType = $('iconType').value;
  const strategy = $('cacheStrategy').value;
  const updateMode = $('updateMode').value;
  const version = TS();

  // Manifest (usa SEMPRE ícones padrão — o favicon detectado é usado no index)
  const manifest = buildManifest({
    title, shortName, description, lang, themeColor, backgroundColor,
    iconsPath, iconType
  });

  const sw = buildSW({strategy, updateMode, version});

  // Index modificado
  const indexMod = injectPWAIntoHTML(raw, {
    title, themeColor, backgroundColor, shortName,
    appleIconHref: faviconHref, useAppleIcon: !!faviconHref
  });

  // Saídas
  $('outManifest').textContent = JSON.stringify(manifest, null, 2);
  $('outSW').textContent = sw;
  $('outIndex').textContent = indexMod;
  return { manifest, sw, indexMod, version, faviconHref };
}

$('btnGenerate').addEventListener('click', generate);

$('btnZip').addEventListener('click', function(){
  const res = generate();
  if (!res) return;
  const zip = new JSZip();
  zip.file('manifest.webmanifest', JSON.stringify(res.manifest, null, 2));
  zip.file('service-worker.js', res.sw);
  zip.file('index.html', res.indexMod);
  const readme = `PWA gerado automaticamente

Arquivos:
- manifest.webmanifest
- service-worker.js
- index.html (modificado com manifest, metas, favicon, banner de update e registro do SW)

Observações:
- Ícones esperados (coloque na pasta): icons/icon-48.png, 72, 96, 128, 144, 152, 192, 256, 384, 512
- Favicon detectado no HTML original: ${res.faviconHref || 'N/D'}
- Estratégia de cache: ${$('cacheStrategy').value}
- Modo de atualização: ${$('updateMode').value}
- Versão: ${res.version}

Dica:
- Navegações são sempre network-first para que seu app atualize rápido.
- Se já existirem tags de manifest/ícones no HTML original, eu substituí pelas necessárias do PWA.
`;
  zip.file('README-PWA.txt', readme);

  zip.generateAsync({type:'blob'}).then(blob=>{
    saveAs(blob, 'pwa-gerado.zip');
  });
});
